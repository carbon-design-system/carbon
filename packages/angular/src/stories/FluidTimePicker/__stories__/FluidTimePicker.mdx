## Overview

The `cds-ng-fluid-time-picker` component provides a fluid variant of the time
picker input, allowing users to enter time values and select time zones.

### Importing

```ts
import {
  FluidTimePickerComponent,
  FluidTimePickerSelectComponent,
  FluidTimePickerSkeletonComponent,
} from '@carbon/angular/fluid-time-picker';
import { SelectItemComponent } from '@carbon/angular/select';

@Component({
  standalone: true,
  imports: [
    FluidTimePickerComponent,
    FluidTimePickerSelectComponent,
    SelectItemComponent,
  ],
  template: `
    <cds-ng-fluid-time-picker labelText="Time" placeholder="hh:mm">
      <cds-ng-fluid-time-picker-select defaultValue="am">
        <cds-ng-select-item value="am" [selected]="true">AM</cds-ng-select-item>
        <cds-ng-select-item value="pm">PM</cds-ng-select-item>
      </cds-ng-fluid-time-picker-select>
    </cds-ng-fluid-time-picker>
  `,
})
export class MyComponent {}
```

## Angular-specific notes

### Forms integration (`ControlValueAccessor`)

`FluidTimePickerComponent` implements `ControlValueAccessor`, supporting both
template-driven and reactive Angular forms:

```html
<!-- Reactive forms -->
<cds-ng-fluid-time-picker [formControl]="timeControl" labelText="Time">
</cds-ng-fluid-time-picker>

<!-- Template-driven forms -->
<cds-ng-fluid-time-picker [(ngModel)]="time" labelText="Time">
</cds-ng-fluid-time-picker>
```
