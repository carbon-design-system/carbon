## Overview

The UI Shell is the shared navigation framework for Carbon. It ties products
together with consistent, reliable topmost navigation and side navigation.

### Importing

```ts
import {
  HeaderComponent,
  HeaderGlobalActionComponent,
  HeaderMenuButtonComponent,
  HeaderNameComponent,
  HeaderNavComponent,
  HeaderNavItemComponent,
  HeaderPanelComponent,
  SideNavComponent,
  SideNavItemsComponent,
  SideNavLinkComponent,
  SideNavMenuComponent,
  SideNavMenuItemComponent,
} from '@carbon/angular/ui-shell';

@Component({
  standalone: true,
  imports: [
    HeaderComponent,
    HeaderGlobalActionComponent,
    HeaderMenuButtonComponent,
    HeaderNameComponent,
    HeaderNavComponent,
    HeaderNavItemComponent,
    HeaderPanelComponent,
    SideNavComponent,
    SideNavItemsComponent,
    SideNavLinkComponent,
    SideNavMenuComponent,
    SideNavMenuItemComponent,
  ],
  template: `
    <cds-ng-header aria-label="IBM Cloud">
      <cds-ng-header-name href="/" prefix="IBM"
        >Cloud console</cds-ng-header-name
      >
      <cds-ng-header-nav menuBarLabel="IBM Cloud">
        <cds-ng-header-nav-item href="/link1">Link 1</cds-ng-header-nav-item>
        <cds-ng-header-nav-item href="/link2">Link 2</cds-ng-header-nav-item>
      </cds-ng-header-nav>
    </cds-ng-header>
  `,
})
export class MyComponent {}
```

## Angular-specific notes

- All sub-components (`HeaderComponent`, `SideNavComponent`,
  `SwitcherComponent`, etc.) must be imported individually from
  `@carbon/angular/ui-shell`.
- Selectors use the `cds-ng-*` prefix (`cds-ng-header`, `cds-ng-side-nav`, etc.)
  to ensure encapsulation and coexistence with other custom elements.
- For icons in slots, use SVG string projection or Carbon icon helpers with
  `slot="icon"` or `slot="title-icon"`.
