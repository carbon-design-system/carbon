## Overview

The `cds-ng-fluid-multi-select` component provides a fluid variant of
multi-select dropdown, allowing users to select multiple options from a list.

### Importing

```ts
import {
  FluidMultiSelectComponent,
  FluidMultiSelectSkeletonComponent,
} from '@carbon/angular/fluid-multi-select';
import { MultiSelectItemComponent } from '@carbon/angular/multi-select';

@Component({
  standalone: true,
  imports: [FluidMultiSelectComponent, MultiSelectItemComponent],
  template: `
    <cds-ng-fluid-multi-select titleText="Label" label="Choose options">
      <cds-ng-multi-select-item value="opt-1"
        >Option 1</cds-ng-multi-select-item
      >
      <cds-ng-multi-select-item value="opt-2"
        >Option 2</cds-ng-multi-select-item
      >
    </cds-ng-fluid-multi-select>
  `,
})
export class MyComponent {}
```

## Angular-specific notes

### Forms integration (`ControlValueAccessor`)

`FluidMultiSelectComponent` implements `ControlValueAccessor`, supporting both
template-driven and reactive Angular forms:

```html
<!-- Reactive forms -->
<cds-ng-fluid-multi-select
  [formControl]="multiControl"
  titleText="Multi-select">
  <cds-ng-multi-select-item value="opt-1">Option 1</cds-ng-multi-select-item>
</cds-ng-fluid-multi-select>

<!-- Template-driven forms -->
<cds-ng-fluid-multi-select
  [(ngModel)]="selectedValues"
  titleText="Multi-select">
  <cds-ng-multi-select-item value="opt-1">Option 1</cds-ng-multi-select-item>
</cds-ng-fluid-multi-select>
```
