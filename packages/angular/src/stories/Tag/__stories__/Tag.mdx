## Overview

Tags are used for items that need to be labeled, categorized, or organized using
keywords that describe them.

### Importing

```ts
import {
  TagComponent,
  DismissibleTagComponent,
  SelectableTagComponent,
  OperationalTagComponent,
  TagSkeletonComponent,
} from '@carbon/angular/tag';

@Component({
  standalone: true,
  imports: [TagComponent, DismissibleTagComponent, SelectableTagComponent],
  template: `
    <cds-ng-tag type="blue">Blue Tag</cds-ng-tag>
    <cds-ng-dismissible-tag
      type="magenta"
      text="Dismissible Tag"></cds-ng-dismissible-tag>
    <cds-ng-selectable-tag text="Selectable Tag"></cds-ng-selectable-tag>
  `,
})
export class MyComponent {}
```
