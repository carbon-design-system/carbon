## Overview

The `cds-ng-fluid-combo-box` component is a fluid variant of the ComboBox,
spanning the full width of its container with specialized styling.

### Importing

```ts
import {
  FluidComboBoxComponent,
  FluidComboBoxSkeletonComponent,
} from '@carbon/angular/fluid-combo-box';
import { ComboBoxItemComponent } from '@carbon/angular/combo-box';

@Component({
  standalone: true,
  imports: [FluidComboBoxComponent, ComboBoxItemComponent],
  template: `
    <cds-ng-fluid-combo-box titleText="Label" label="Choose an option">
      <cds-ng-combo-box-item value="option-0">Option 0</cds-ng-combo-box-item>
      <cds-ng-combo-box-item value="option-1">Option 1</cds-ng-combo-box-item>
    </cds-ng-fluid-combo-box>
  `,
})
export class MyComponent {}
```

## Angular-specific notes

### Forms integration (`ControlValueAccessor`)

`FluidComboBoxComponent` implements `ControlValueAccessor`, supporting both
template-driven and reactive Angular forms:

```html
<!-- Reactive forms -->
<cds-ng-fluid-combo-box [formControl]="comboControl" titleText="Combo box">
  <cds-ng-combo-box-item value="opt-1">Option 1</cds-ng-combo-box-item>
</cds-ng-fluid-combo-box>

<!-- Template-driven forms -->
<cds-ng-fluid-combo-box [(ngModel)]="selectedValue" titleText="Combo box">
  <cds-ng-combo-box-item value="opt-1">Option 1</cds-ng-combo-box-item>
</cds-ng-fluid-combo-box>
```
