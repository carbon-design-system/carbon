## Overview

The `cds-ng-fluid-select` component provides a fluid variant of the select
input, spanning the full width of its container.

### Importing

```ts
import {
  FluidSelectComponent,
  FluidSelectSkeletonComponent,
} from '@carbon/angular/fluid-select';
import { SelectItemComponent } from '@carbon/angular/select';

@Component({
  standalone: true,
  imports: [FluidSelectComponent, SelectItemComponent],
  template: `
    <cds-ng-fluid-select labelText="Select an option">
      <cds-ng-select-item value="opt-1">Option 1</cds-ng-select-item>
      <cds-ng-select-item value="opt-2">Option 2</cds-ng-select-item>
    </cds-ng-fluid-select>
  `,
})
export class MyComponent {}
```

## Angular-specific notes

### Forms integration (`ControlValueAccessor`)

`FluidSelectComponent` implements `ControlValueAccessor`, supporting both
template-driven and reactive Angular forms:

```html
<!-- Reactive forms -->
<cds-ng-fluid-select [formControl]="selectControl" labelText="Region">
  <cds-ng-select-item value="us-south">Dallas</cds-ng-select-item>
  <cds-ng-select-item value="eu-de">Frankfurt</cds-ng-select-item>
</cds-ng-fluid-select>

<!-- Template-driven forms -->
<cds-ng-fluid-select [(ngModel)]="selectedRegion" labelText="Region">
  <cds-ng-select-item value="us-south">Dallas</cds-ng-select-item>
  <cds-ng-select-item value="eu-de">Frankfurt</cds-ng-select-item>
</cds-ng-fluid-select>
```
