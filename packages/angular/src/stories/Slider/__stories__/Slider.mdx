## Overview

The `cds-ng-slider` component allows users to choose from a range of values by
moving a thumb along a horizontal track.

### Importing

```ts
import {
  SliderComponent,
  SliderInputComponent,
  SliderSkeletonComponent,
} from '@carbon/angular/slider';
import { FormItemComponent } from '@carbon/angular/form';

@Component({
  standalone: true,
  imports: [SliderComponent, SliderInputComponent, FormItemComponent],
  template: `
    <cds-ng-form-item>
      <cds-ng-slider
        labelText="Storage allocation"
        [min]="0"
        [max]="100"
        [step]="1"
        [value]="50">
        <cds-ng-slider-input aria-label="Slider value" type="number">
        </cds-ng-slider-input>
      </cds-ng-slider>
    </cds-ng-form-item>
  `,
})
export class MyComponent {}
```

## Angular-specific notes

### Forms integration (`ControlValueAccessor`)

`SliderComponent` implements `ControlValueAccessor`, supporting both
template-driven and reactive Angular forms:

```html
<!-- Reactive forms -->
<cds-ng-form-item>
  <cds-ng-slider [formControl]="storageControl" labelText="Storage allocation">
    <cds-ng-slider-input
      aria-label="Slider value"
      type="number"></cds-ng-slider-input>
  </cds-ng-slider>
</cds-ng-form-item>

<!-- Template-driven forms -->
<cds-ng-form-item>
  <cds-ng-slider [(ngModel)]="storageAllocation" labelText="Storage allocation">
    <cds-ng-slider-input
      aria-label="Slider value"
      type="number"></cds-ng-slider-input>
  </cds-ng-slider>
</cds-ng-form-item>
```
